(dict "initial" $initial "posts" $posts "terms" $terms "url" (printf "%s#term-%s" $base .id) "sortKey" (printf "%s-%s" $initial.key (lower .title)))) -}} {{- end -}} {{- end -}} {{- return (sort $entries "sortKey") -}}
